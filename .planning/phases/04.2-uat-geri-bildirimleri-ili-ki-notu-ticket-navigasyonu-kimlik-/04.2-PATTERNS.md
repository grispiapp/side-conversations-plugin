const closeMenu = useCallback((returnFocus = true) => {

```typescript
// Source: src/lib/last-seen-store.ts (existing, this repo)
const KEY_PREFIX = "sc:infoBoxSeen:"; // new prefix, same namespace convention

function storageKey(tenantId: string): string {
  return `${KEY_PREFIX}${encodeURIComponent(tenantId)}`;
}

export function getInfoBoxSeen(tenantId: string): boolean {
  if (!tenantId.trim()) return false;
  try {
    return window.localStorage.getItem(storageKey(tenantId)) === "1";
  } catch {
    return false; // defensive degrade — matches last-seen-store's own comment on iframe storage restrictions throwing on ACCESS, not just write
  }
}

export function setInfoBoxSeen(tenantId: string): boolean {
  if (!tenantId.trim()) return false;
  try {
    window.localStorage.setItem(storageKey(tenantId), "1");
    return true;
  } catch {
    return false;
  }
}
```
Note `last-seen-store.ts`'s real functions take BOTH `tenantId` and `ticketKey` (`storageKey(tenantId, ticketKey)`, lines 3-5) since it's per-conversation — `info-box-store.ts` deliberately drops the ticket-key dimension per D-16's own "tenant bazlı" wording. Defensive `.trim()` guard and try/catch-degrades-to-safe-default are the two load-bearing details to preserve exactly.
